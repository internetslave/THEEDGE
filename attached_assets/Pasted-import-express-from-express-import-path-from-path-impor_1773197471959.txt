function loadJSON(file, def) {
  try { return JSON.parse(fs.readFileSync(file, 'utf8')); }
  catch { return def; }
}
function saveJSON(file, data) {
  fs.writeFileSync(file, JSON.stringify(data, null, 2), 'utf8');
}
function isFresh(cache, key, ttl) {
  return cache[key] && (Date.now() - cache[key].ts < ttl);
}

const aiCache = {};

function loadAICache() {
  try {
    const data = JSON.parse(fs.readFileSync(CACHE_FILE, 'utf8'));
    const now  = Date.now();
    let loaded = 0;
    for (const [k, v] of Object.entries(data)) {
      if (now - v.ts < AI_TTL) { aiCache[k] = v; loaded++; }
    }
    console.log(`📦 Loaded ${loaded} cached AI analyses from disk`);
  } catch { console.log('No AI cache yet — starting fresh'); }
}
function saveAICache() {
  try { fs.writeFileSync(CACHE_FILE, JSON.stringify(aiCache), 'utf8'); }
  catch (e) { console.error('Failed to save AI cache:', e.message); }
}
loadAICache();

function hashPin(pin, salt)  { return crypto.pbkdf2Sync(pin, salt, 10000, 32, 'sha256').toString('hex'); }
function genSalt()           { return crypto.randomBytes(16).toString('hex'); }
function genToken()          { return crypto.randomBytes(32).toString('hex'); }

function createSession(username) {
  const token = genToken();
  sessions[token] = { username, expires: Date.now() + SESSION_TTL };
  return token;
}
function getSession(token) {
  if (!token) return null;
  const s = sessions[token];
  if (!s) return null;
  if (Date.now() > s.expires) { delete sessions[token]; return null; }
  return s.username;
}
function authMiddleware(req, res, next) {
  const token    = req.headers['x-session-token'] || req.query.token;
  const username = getSession(token);
  if (!username) return res.status(401).json({ error: 'Not authenticated' });
  req.username = username;
  next();
}

const rateLimitStore = new Map();
const RATE_LIMIT  = 30;
const RATE_WINDOW = 60 * 1000;

function rateLimit(req, res, next) {
  const ip  = req.headers['x-forwarded-for']?.split(',')[0] || req.socket.remoteAddress || 'unknown';
  const now = Date.now();
  const rec = rateLimitStore.get(ip) || { count: 0, start: now };
  if (now - rec.start > RATE_WINDOW) { rec.count = 1; rec.start = now; }
  else rec.count++;
  rateLimitStore.set(ip, rec);
  if (rateLimitStore.size > 1000) {
    for (const [k, v] of rateLimitStore) {
      if (now - v.start > RATE_WINDOW) rateLimitStore.delete(k);
    }
  }
  if (rec.count > RATE_LIMIT) return res.status(429).json({ error: 'Too many requests' });
  next();
}

const oddsCache = {};
const ODDS_TTL  = 5 * 60 * 1000;

async function fetchOdds(sport) {
  if (isFresh(oddsCache, sport, ODDS_TTL)) return oddsCache[sport].data;
  try {
    const url = `${ODDS_BASE}/sports/${sport}/odds/?apiKey=${ODDS_API_KEY}&regions=au&markets=h2h&oddsFormat=decimal&dateFormat=iso`;
    const res = await fetch(url);
    if (!res.ok) throw new Error(`Odds API ${res.status}`);
    const data = await res.json();
    oddsCache[sport] = { data, ts: Date.now() };
    return data;
  } catch (err) {
    console.error(`fetchOdds(${sport}):`, err.message);
    return oddsCache[sport]?.data || [];
  }
}

async function fetchAllOdds() {
  const sportMap = {
    afl: 'aussierules_afl', nrl: 'rugbyleague_nrl',
    soccer_al: 'soccer_australia_aleague', soccer_epl: 'soccer_epl',
    ufc: 'mma_mixed_martial_arts', boxing: 'boxing_boxing',
  };
  const results = {};
  for (const [key, apiSport] of Object.entries(sportMap)) {
    results[key] = await fetchOdds(apiSport);
  }
  return results;
}

function normaliseEvent(ev, sportLabel) {
  const bookmakers = ev.bookmakers || [];
  const bookie = bookmakers.find(b => b.key === 'sportsbet')
    || bookmakers.find(b => b.key === 'tab') || bookmakers[0];
  if (!bookie) return null;
  const market = bookie.markets?.find(m => m.key === 'h2h');
  if (!market || !market.outcomes?.length) return null;

  const outcomes = market.outcomes;
  const home = outcomes[0], away = outcomes[1];
  const draw = outcomes.find(o => o.name === 'Draw');

  const TZ = 'Australia/Brisbane';
  const commence = new Date(ev.commence_time), now = new Date();
  const diffHrs   = (commence - now) / 3600000;
  const toDate    = (d) => d.toLocaleDateString('en-AU', { timeZone: TZ });
  const tomorrow  = new Date(now); tomorrow.setDate(tomorrow.getDate() + 1);
  const timeStr   = commence.toLocaleTimeString('en-AU', { hour: '2-digit', minute: '2-digit', timeZone: TZ });
  const days      = ['Sun','Mon','Tue','Wed','Thu','Fri','Sat'];
  const dayBne    = new Date(commence.toLocaleString('en-AU', { timeZone: TZ })).getDay();

  let timeLabel;
  if (diffHrs < 0)                        timeLabel = 'In Progress';
  else if (diffHrs < 1)                   timeLabel = `${Math.round(diffHrs * 60)}m`;
  else if (toDate(commence) === toDate(now))      timeLabel = `Today ${timeStr}`;
  else if (toDate(commence) === toDate(tomorrow)) timeLabel = `Tomorrow ${timeStr}`;
  else                                    timeLabel = `${days[dayBne]} ${timeStr}`;

  const homeImpl = home ? 1/home.price : 0, awayImpl = away ? 1/away.price : 0;
  const overround = homeImpl + awayImpl + (draw ? 1/draw.price : 0);
  const homeEdge  = home ? (home.price * homeImpl/overround - 1) : -1;
  const awayEdge  = away ? (away.price * awayImpl/overround - 1) : -1;

  return {
    id: ev.id, sport: sportLabel,
    home: home?.name || ev.home_team, away: away?.name || ev.away_team,
    homeOdds: home?.price ? parseFloat(home.price.toFixed(2)) : null,
    awayOdds: away?.price ? parseFloat(away.price.toFixed(2)) : null,
    drawOdds: draw?.price ? parseFloat(draw.price.toFixed(2)) : null,
    time: timeLabel, commenceTime: ev.commence_time,
    venue: '', comp: ev.sport_title || sportLabel,
    homeRecord: '', awayRecord: '', homeEdge, awayEdge, bookmaker: bookie.title,
  };
}

async function analyseMatch(ev) {
  if (isFresh(aiCache, ev.id, AI_TTL)) return aiCache[ev.id].data;
  if (!ANTHROPIC_API_KEY) return null;

  const prompt = `You are an expert sports betting analyst with deep knowledge of Australian sports.
Analyse this ${ev.sport} match and provide a betting recommendation.
MATCH: ${ev.home} vs ${ev.away}
COMPETITION: ${ev.comp}
ODDS: ${ev.home} @ ${ev.homeOdds} | ${ev.away} @ ${ev.awayOdds}${ev.drawOdds ? ` | Draw @ ${ev.drawOdds}` : ''}
KICKOFF: ${ev.time}
Respond ONLY with valid JSON, no markdown:
{"confidence":<45-90>,"recommendation":"<must be exactly '${ev.home}' or '${ev.away}'${ev.drawOdds ? " or 'Draw'" : ''}>","betType":"<Win or Head to Head>","recommendedOdds":"<string>","valueBet":<bool>,"summary":"<2 sentences>","keyFactors":[{"icon":"📊","label":"Recent Form","value":"<brief>"},{"icon":"🏟️","label":"Conditions","value":"<brief>"},{"icon":"📈","label":"Head to Head","value":"<brief>"},{"icon":"💰","label":"Market Value","value":"<brief>"},{"icon":"⚠️","label":"Key Risk","value":"<brief>"}],"reasoning":"<2-3 sentences>"}`;

  try {
    const res = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-api-key': ANTHROPIC_API_KEY, 'anthropic-version': '2023-06-01' },
      body: JSON.stringify({ model: 'claude-sonnet-4-20250514', max_tokens: 800, messages: [{ role: 'user', content: prompt }] }),
    });
    if (!res.ok) throw new Error(`Anthropic ${res.status}`);
    const data     = await res.json();
    const analysis = JSON.parse(data.content?.[0]?.text?.replace(/```json|```/g,'').trim() || '{}');
    aiCache[ev.id] = { data: analysis, ts: Date.now() };
    saveAICache();
    console.log(`✅ AI: ${ev.home} vs ${ev.away} → ${analysis.recommendation} (${analysis.confidence}%)`);
    return analysis;
  } catch (err) {
    console.error(`❌ AI failed ${ev.home} vs ${ev.away}:`, err.message);
    return null;
  }
}

async function processAIQueue() {
  if (aiBusy || aiQueue.length === 0) return;
  aiBusy = true;
  while (aiQueue.length > 0) {
    const ev = aiQueue.shift();
    if (!isFresh(aiCache, ev.id, AI_TTL)) {
      await analyseMatch(ev);
      await new Promise(r => setTimeout(r, 800));
    }
  }
  aiBusy = false;
}

function queueAIAnalysis(events) {
  if (!ANTHROPIC_API_KEY) return;
  const toAnalyse = events.filter(ev => !isFresh(aiCache, ev.id, AI_TTL)).slice(0, 10);
  for (const ev of toAnalyse) {
    if (!aiQueue.find(q => q.id === ev.id)) aiQueue.push(ev);
  }
  processAIQueue();
}

app.post('/api/auth/signup', (req, res) => {
  let { username, pin, sport } = req.body;
  if (!username || !pin) return res.status(400).json({ error: 'Username and PIN required' });
  username = username.toLowerCase().trim();
  if (username.length < 2 || username.length > 20) return res.status(400).json({ error: 'Username must be 2-20 characters' });
  if (!/^[a-z0-9_]+$/.test(username)) return res.status(400).json({ error: 'Letters, numbers and underscores only' });
  if (!/^\d{4}$/.test(pin)) return res.status(400).json({ error: 'PIN must be exactly 4 digits' });
  if (users[username]) return res.status(409).json({ error: 'Username already taken' });
  const salt = genSalt(), pinHash = hashPin(pin, salt);
  const idx  = Object.keys(users).length % AVATARS.length;
  users[username]  = { pinHash, salt, sport: sport || 'AFL', avatar: AVATARS[idx], color: COLORS[idx], createdAt: Date.now() };
  betsDB[username] = [];
  saveJSON(USERS_FILE, users);
  saveJSON(BETS_FILE, betsDB);
  const token = createSession(username);
  res.json({ success: true, token, profile: { username, avatar: users[username].avatar, color: users[username].color, sport: users[username].sport } });
});

app.post('/api/auth/signin', (req, res) => {
  let { username, pin } = req.body;
  if (!username || !pin) return res.status(400).json({ error: 'Username and PIN required' });
  username = username.toLowerCase().trim();
  const user = users[username];
  if (!user) return res.status(404).json({ error: 'Account not found' });
  if (hashPin(pin, user.salt) !== user.pinHash) return res.status(401).json({ error: 'Incorrect PIN' });
  const token = createSession(username);
  res.json({ success: true, token, profile: { username, avatar: user.avatar, color: user.color, sport: user.sport } });
});

app.get('/api/odds', rateLimit, async (req, res) => {
  try {
    const raw = await fetchAllOdds();
    const sportLabels = { afl:'AFL', nrl:'NRL', soccer_al:'Soccer', soccer_epl:'Soccer', ufc:'UFC', boxing:'Boxing' };
    const events = [];
    for (const [key, evList] of Object.entries(raw)) {
      for (const ev of (evList || [])) {
        const norm = normaliseEvent(ev, sportLabels[key]);
        if (!norm) continue;
        const ai = aiCache[norm.id]?.data || null;
        if (ai) {
          norm.confidence = ai.confidence; norm.pick = ai.recommendation;
          norm.valueBet = ai.valueBet; norm.aiAnalysis = ai; norm.aiReady = true;
        } else {
          norm.confidence = Math.min(95, Math.max(45, Math.round(50 + Math.max(norm.homeEdge, norm.awayEdge) * 100)));
          norm.pick = norm.homeEdge >= norm.awayEdge ? norm.home : norm.away;
          norm.valueBet = Math.max(norm.homeEdge, norm.awayEdge) > 0.03;
          norm.aiReady = false;
        }
        events.push(norm);
      }
    }
    events.sort((a, b) => new Date(a.commenceTime) - new Date(b.commenceTime));
    queueAIAnalysis(events);
    res.json({ success: true, events, count: events.length });
  } catch (err) {
    console.error('Odds error:', err.message);
    res.status(500).json({ success: false, error: 'Failed to load odds' });
  }
});

// Scoring formula:
//   Base: correct pick = 1pt
//   Confidence bonus: confidence rank (1-8) * 0.5 if correct (so max 4 extra pts)
//   Margin bonus: if margin tip within 12 pts = +1, within 6 pts = +2, exact = +5
//   Banker: if correct = x2 all points for that game
function scoreOnePick(pick, result) {
  if (!result || result.status !== 'final') return null;
  let pts = 0;
  const correct = pick.tip === result.winner;
  if (correct) {
    pts += 1;
    // confidence bonus
    if (pick.confidence) pts += Math.round((pick.confidence / 8) * 4 * 10) / 10;
    // margin bonus
    if (pick.margin != null && result.margin != null) {
      const diff = Math.abs(pick.margin - result.margin);
      if (diff === 0)       pts += 5;
      else if (diff <= 6)   pts += 2;
      else if (diff <= 12)  pts += 1;
    }
  }
  if (pick.banker) pts *= 2;
  return { pts: Math.round(pts * 10) / 10, correct, pick: pick.tip, actual: result.winner };
}

// POST /api/tipping/submit — submit tips for current round
app.post('/api/tipping/submit', authMiddleware, (req, res) => {
  const { tips } = req.body; // { fixtureId: { tip, margin, confidence, banker } }
  if (!tips || typeof tips !== 'object') return res.status(400).json({ error: 'Invalid tips' });
  const tipping = loadTipping();
  const round = tipping.activeRound;
  if (!round) return res.status(400).json({ error: 'No active round' });

  const fixtures = tipping.rounds[round]?.fixtures || [];
  const lockout = tipping.rounds[round]?.lockout;
  if (lockout && new Date() > new Date(lockout)) {
    return res.status(400).json({ error: 'Round has locked out' });
  }

  // Validate tips against fixtures
  const validIds = new Set(fixtures.map(f => f.id));
  const cleanTips = {};
  for (const [id, t] of Object.entries(tips)) {
    if (!validIds.has(id)) continue;
    if (!t.tip) continue;
    cleanTips[id] = {
      tip: String(t.tip).substring(0, 100),
      margin: t.margin != null ? Math.max(0, Math.min(200, parseInt(t.margin) || 0)) : null,
      confidence: t.confidence ? Math.max(1, Math.min(8, parseInt(t.confidence))) : null,
      banker: !!t.banker,
      submittedAt: new Date().toISOString()
    };
  }

  if (!tipping.results[round]) tipping.results[round] = {};
  tipping.results[round][req.username] = cleanTips;
  saveTipping(tipping);
  res.json({ success: true, tipsCount: Object.keys(cleanTips).length });
});

app.get('/api/tipping/leaderboard', (req, res) => {
  const tipping = loadTipping();
  const users = loadJSON(USERS_FILE, {});
  const scores = {}; // { username: { total, rounds: {}, correct, tips } }

  for (const [round, roundResults] of Object.entries(tipping.results)) {
    const fixtures = tipping.rounds[round]?.fixtures || [];
    const results  = tipping.rounds[round]?.results  || {};
    for (const [username, tips] of Object.entries(roundResults)) {
      if (!scores[username]) scores[username] = { total: 0, correct: 0, tips: 0, rounds: {} };
      let roundPts = 0;
      let roundCorrect = 0;
      for (const [fid, pick] of Object.entries(tips)) {
        const result = results[fid];
        if (!result) continue;
        const scored = scoreOnePick(pick, result);
        if (scored) {
          roundPts += scored.pts;
          if (scored.correct) roundCorrect++;
          scores[username].tips++;
        }
      }
      scores[username].total += roundPts;
      scores[username].correct += roundCorrect;
      scores[username].rounds[round] = { pts: roundPts, correct: roundCorrect };
    }
  }

  const board = Object.entries(scores).map(([username, s]) => ({
    username,
    avatar: users[username]?.avatar || '🎯',
    color:  users[username]?.color  || '#aaa',
    total:  Math.round(s.total * 10) / 10,
    correct: s.correct,
    tips:    s.tips,
    winRate: s.tips ? Math.round((s.correct / s.tips) * 100) : 0,
    rounds:  s.rounds,
  })).sort((a, b) => b.total - a.total);

  res.json({ board, activeRound: tipping.activeRound });
});

// GET /api/tipping/comps/mine
app.get('/api/tipping/comps/mine', authMiddleware, (req, res) => {
  const comps = loadComps();
  const tipping = loadTipping();
  const codes = comps.memberships[req.username] || [];
  const users = loadJSON(USERS_FILE, {});

  const myComps = codes.map(code => {
    const c = comps.comps[code];
    if (!c) return null;
    // Build mini leaderboard for this comp
    const scores = {};
    for (const [round, roundResults] of Object.entries(tipping.results)) {
      const results = tipping.rounds[round]?.results || {};
      for (const member of c.members) {
        const tips = roundResults[member] || {};
        if (!scores[member]) scores[member] = { total: 0, correct: 0, tips: 0 };
        for (const [fid, pick] of Object.entries(tips)) {
          const result = results[fid];
          if (!result) continue;
          const scored = scoreOnePick(pick, result);
          if (scored) {
            scores[member].total += scored.pts;
            if (scored.correct) scores[member].correct++;
            scores[member].tips++;
          }
        }
      }
    }
    const board = c.members.map(m => ({
      username: m, avatar: users[m]?.avatar || '🎯', color: users[m]?.color || '#aaa',
      total: Math.round((scores[m]?.total || 0) * 10) / 10,
      correct: scores[m]?.correct || 0, tips: scores[m]?.tips || 0,
    })).sort((a, b) => b.total - a.total);
    return { ...c, board };
  }).filter(Boolean);

  res.json({ comps: myComps });
});

  res.json({ success: true });
});
