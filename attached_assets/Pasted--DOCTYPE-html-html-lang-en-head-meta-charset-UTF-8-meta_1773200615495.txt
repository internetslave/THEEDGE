// ═══════════════════════════════════════════════════════════════════
// STATE
// ═══════════════════════════════════════════════════════════════════
let tipToken    = localStorage.getItem('edgeiq-token-v2');
let currentUser = null;
let fixtures    = [];
let myTips      = {}; // { fixtureId: { tip, margin, confidence, banker } }
let activeRound = null;
let isLocked    = false;
let roundLockout = null;

const SPORT_BADGES = {
  AFL:    { emoji:'🏉', cls:'sport-afl',    label:'AFL'    },
  NRL:    { emoji:'🏈', cls:'sport-nrl',    label:'NRL'    },
  EPL:    { emoji:'⚽', cls:'sport-epl',    label:'EPL'    },
  Origin: { emoji:'🔥', cls:'sport-origin', label:'State of Origin' },
};

// ═══════════════════════════════════════════════════════════════════
// AUTH
// ═══════════════════════════════════════════════════════════════════
function showAuthTab(tab, btn) {
  document.querySelectorAll('.auth-tab').forEach(b => b.classList.remove('active'));
  if (btn) btn.classList.add('active');
  document.getElementById('auth-in').style.display = tab === 'in' ? 'block' : 'none';
  document.getElementById('auth-up').style.display = tab === 'up' ? 'block' : 'none';
}

async function doTipSignIn() {
  const user = document.getElementById('ai-user').value.trim();
  const pin  = document.getElementById('ai-pin').value.trim();
  if (!user || !pin) return showAuthError('Enter username and PIN');
  try {
    const r = await fetch('/api/auth/signin', {
      method:'POST', headers:{'Content-Type':'application/json'},
      body: JSON.stringify({ username: user, pin })
    });
    const d = await r.json();
    if (!d.token) return showAuthError(d.error || 'Sign in failed');
    tipToken = d.token;
    localStorage.setItem('edgeiq-token-v2', tipToken);
    currentUser = d.profile;
    onLoggedIn();
  } catch { showAuthError('Connection error — try again'); }
}

async function doTipSignUp() {
  const user = document.getElementById('au-user').value.trim();
  const pin  = document.getElementById('au-pin').value.trim();
  const pin2 = document.getElementById('au-pin2').value.trim();
  if (!user || !pin) return showAuthError('Fill in all fields');
  if (pin !== pin2)  return showAuthError('PINs don\'t match');
  if (pin.length !== 4 || !/^\d+$/.test(pin)) return showAuthError('PIN must be exactly 4 digits');
  try {
    const r = await fetch('/api/auth/signup', {
      method:'POST', headers:{'Content-Type':'application/json'},
      body: JSON.stringify({ username: user, pin, sport: 'AFL' })
    });
    const d = await r.json();
    if (!d.token) return showAuthError(d.error || 'Sign up failed');
    tipToken = d.token;
    localStorage.setItem('edgeiq-token-v2', tipToken);
    currentUser = d.profile;
    onLoggedIn();
  } catch { showAuthError('Connection error — try again'); }
}

function showAuthError(msg) {
  const el = document.getElementById('tip-auth-error');
  el.textContent = msg;
  el.style.display = 'block';
}

async function checkAuth() {
  if (!tipToken) return;
  try {
    const r = await fetch('/api/auth/me', { headers: { 'x-session-token': tipToken } });
    if (!r.ok) return;
    const d = await r.json();
    // /api/auth/me returns { username, avatar, color, sport } directly — no wrapper
    if (d.username) {
      currentUser = { username: d.username, avatar: d.avatar, color: d.color, sport: d.sport };
      onLoggedIn();
    }
  } catch {}
}

function onLoggedIn() {
  document.getElementById('tip-auth').style.display = 'none';
  const badge = document.getElementById('tip-user-badge');
  badge.style.display = 'flex';
  badge.innerHTML = `
    <span style="font-size:18px">${currentUser.avatar}</span>
    <span style="font-family:var(--head);font-size:13px;font-weight:700;letter-spacing:.5px">${currentUser.username.toUpperCase()}</span>`;
  loadAll();
}

// ═══════════════════════════════════════════════════════════════════
// NAVIGATION
// ═══════════════════════════════════════════════════════════════════
function showTipPage(page, btn) {
  document.querySelectorAll('.tip-page').forEach(p => p.classList.remove('active'));
  document.getElementById('tip-page-' + page).classList.add('active');
  document.querySelectorAll('.tip-nav-btn').forEach(b => b.classList.remove('active'));
  if (btn) btn.classList.add('active');
  document.getElementById('submit-bar').style.display = page === 'tips' && !isLocked ? 'flex' : 'none';
  window.scrollTo({ top: 0, behavior: 'smooth' });
  if (page === 'leaderboard') loadLeaderboard();
  if (page === 'comps') loadComps();
  if (page === 'results') loadResultsRounds();
}

// ═══════════════════════════════════════════════════════════════════
// LOAD ALL
// ═══════════════════════════════════════════════════════════════════
async function loadAll() {
  await Promise.all([loadFixtures(), loadMyTips()]);
}

async function loadFixtures() {
  try {
    const r = await fetch('/api/tipping/fixtures');
    console.log('[fixtures] status:', r.status);
    const d = await r.json();
    console.log('[fixtures] response:', JSON.stringify(d));
    activeRound   = d.round;
    fixtures      = d.fixtures || [];
    roundLockout  = d.lockout  || null;
    renderFixtures();
  } catch(e) {
    console.error('[fixtures] failed:', e);
    renderFixtures();
  }
}

async function loadMyTips() {
  if (!tipToken) return;
  try {
    const r = await fetch('/api/tipping/my-tips', { headers: { 'x-session-token': tipToken } });
    const d = await r.json();
    myTips = d.tips || {};
    updateProgress();
  } catch {}
}

// ═══════════════════════════════════════════════════════════════════
// RENDER FIXTURES
// ═══════════════════════════════════════════════════════════════════
function renderFixtures() {
  const list  = document.getElementById('fixtures-list');
  const empty = document.getElementById('fixtures-empty');

  if (!activeRound || !fixtures.length) {
    list.innerHTML = '';
    empty.style.display = 'block';
    document.getElementById('tips-round-title').textContent = 'No Active Round';
    document.getElementById('submit-bar').style.display = 'none';
    return;
  }
  empty.style.display = 'none';

  document.getElementById('tips-round-title').textContent = activeRound;
  document.getElementById('tips-round-meta').textContent = `${fixtures.length} matches to tip`;

  // Check lockout
  const lockoutTime = roundLockout;
  isLocked = lockoutTime ? new Date() > new Date(lockoutTime) : false;
  const lockBadge = document.getElementById('tips-lockout-badge');
  if (lockoutTime) {
    const d = new Date(lockoutTime);
    const fmt = d.toLocaleDateString('en-AU',{weekday:'short',month:'short',day:'numeric'}) + ' ' + d.toLocaleTimeString('en-AU',{hour:'2-digit',minute:'2-digit'});
    lockBadge.innerHTML = isLocked
      ? `<div class="lockout-badge locked">🔒 LOCKED</div>`
      : `<div class="lockout-badge">⏰ LOCKOUT ${fmt}</div>`;
  }

  document.getElementById('submit-bar').style.display = isLocked ? 'none' : 'flex';

  list.innerHTML = fixtures.map(f => renderFixtureCard(f)).join('');
  updateProgress();
}

          <div class="tip-btn-check">✓</div>

    <div style="padding:0 20px 16px">
      ${saved.tip
        ? `<div style="background:var(--lime-dim);border:1px solid rgba(184,255,60,.2);border-radius:8px;padding:12px 16px;display:flex;align-items:center;justify-content:space-between">

            ${saved.confidence ? `<div style="font-family:var(--mono);font-size:12px;color:var(--muted)">CONF: ${saved.confidence}/8</div>` : ''}
          </div>`
        : `<div style="background:rgba(255,60,90,.08);border:1px solid rgba(255,60,90,.2);border-radius:8px;padding:12px 16px;font-family:var(--head);font-size:13px;color:var(--red)">⚠️ NO TIP SUBMITTED — ROUND LOCKED</div>`
      }
    </div>`}
  </div>`;
}

function selectTip(fixtureId, team) {
  if (isLocked) return;
  if (!myTips[fixtureId]) myTips[fixtureId] = {};
  myTips[fixtureId].tip = team;
  // Re-render just this card
  const card = document.getElementById('fc-' + fixtureId);
  const f    = fixtures.find(x => x.id === fixtureId);
  if (card && f) card.outerHTML = renderFixtureCard(f);
  updateProgress();
}

function toggleExtras(fixtureId) {
  const el    = document.getElementById('extras-' + fixtureId);
  const arrow = document.getElementById('extras-arrow-' + fixtureId);
  const open  = el.classList.toggle('open');
  if (arrow) arrow.textContent = open ? '▾' : '▸';
}

function updateExtra(fixtureId, key, value) {
  if (!myTips[fixtureId]) myTips[fixtureId] = {};
  if (key === 'margin')     myTips[fixtureId].margin     = parseInt(value) || null;
  if (key === 'confidence') {
    myTips[fixtureId].confidence = parseInt(value);
    // Update conf buttons
    document.querySelectorAll(`#extras-${fixtureId} .conf-btn`).forEach((b, i) => {
      b.classList.toggle('selected', i + 1 === parseInt(value));
    });
  }
}

function toggleBanker(fixtureId) {
  if (!myTips[fixtureId]) myTips[fixtureId] = {};
  myTips[fixtureId].banker = !myTips[fixtureId].banker;
  const btn = document.getElementById('banker-' + fixtureId);
  if (btn) {
    btn.classList.toggle('on', myTips[fixtureId].banker);
    btn.innerHTML = `<span>${myTips[fixtureId].banker ? '🌟' : '☆'}</span> ${myTips[fixtureId].banker ? 'BANKER ON' : 'Set as Banker'}`;
  }
}

function updateProgress() {
  const total  = fixtures.length;
  const done   = fixtures.filter(f => myTips[f.id]?.tip).length;
  const pct    = total ? Math.round((done / total) * 100) : 0;
  document.getElementById('progress-done').textContent  = done;
  document.getElementById('progress-total').textContent = total;
  document.getElementById('progress-fill').style.width  = pct + '%';
  document.getElementById('sb-count').textContent = done;
  const btn = document.getElementById('submit-btn-el');
  // Only disable if zero tips AND fixtures exist — always clickable if no fixtures so user gets a message
  if (btn) btn.disabled = total > 0 && done === 0;
}

// SUBMIT
// ═══════════════════════════════════════════════════════════════════
async function submitTips() {
  console.log('[submit] clicked. tipToken:', !!tipToken, '| myTips:', JSON.stringify(myTips));
  if (!tipToken) { showToast('⚠️ Not logged in'); return; }

  const tipsToSend = {};
  for (const [id, t] of Object.entries(myTips)) {
    if (t.tip) tipsToSend[id] = t;
  }
  console.log('[submit] tipsToSend count:', Object.keys(tipsToSend).length);
  if (!Object.keys(tipsToSend).length) { showToast('⚠️ Pick at least one team first'); return; }

  const btn = document.getElementById('submit-btn-el');
  if (btn) { btn.disabled = true; btn.textContent = 'SUBMITTING...'; }

  try {
    const r = await fetch('/api/tipping/submit', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-session-token': tipToken },
      body: JSON.stringify({ tips: tipsToSend })
    });
    console.log('[submit] status:', r.status);
    const d = await r.json();
    console.log('[submit] response:', JSON.stringify(d));
    if (d.success) {
      showToast('✓ ' + d.tipsCount + ' tips submitted!');
      await loadMyTips();
      renderFixtures();
    } else {
      showToast('⚠️ ' + (d.error || 'Submit failed'));
    }
  } catch(e) {
    console.error('[submit] error:', e);
    showToast('⚠️ Connection error');
  } finally {
    if (btn) { btn.disabled = false; btn.textContent = 'SUBMIT TIPS'; }
  }
}

async function loadResultsRounds() {
  // Populate round select from tipping data
  try {
    const r = await fetch('/api/tipping/leaderboard');
    const d = await r.json();
    const select = document.getElementById('results-round-select');
    const rounds = new Set();
    (d.board || []).forEach(e => Object.keys(e.rounds || {}).forEach(rd => rounds.add(rd)));
    if (d.activeRound) rounds.add(d.activeRound);
    select.innerHTML = '<option value="">Select round...</option>' +
      [...rounds].sort().reverse().map(rd => `<option value="${rd}">${rd}</option>`).join('');
    if (rounds.size) { select.value = [...rounds][0]; loadResults(); }
  } catch {}
}

async function loadResults() {
  const round = document.getElementById('results-round-select').value;
  const list  = document.getElementById('results-list');
  const empty = document.getElementById('results-empty');
  if (!round) { list.innerHTML = ''; empty.style.display = 'block'; return; }

  try {
    const r = await fetch(`/api/tipping/round-results?round=${encodeURIComponent(round)}`);
    const d = await r.json();
    const fixturesR = d.fixtures || [];
    const results   = d.results  || {};
    const myRoundTips = (await fetch('/api/tipping/my-tips', { headers: { 'x-session-token': tipToken } }).then(r => r.json())).tips || {};

    let pts = 0, correct = 0, total = 0;
    list.innerHTML = fixturesR.map(f => {
      const result = results[f.id];
      const myPick = myRoundTips[f.id];
      let tag = '', ptsEl = '';
      if (result?.status === 'final' && myPick) {
        const scored = scoreOnePick(myPick, result);
        if (scored) {
          pts += scored.pts; total++;
          if (scored.correct) correct++;
          tag = `<span class="result-pick-tag ${scored.correct ? 'correct' : 'wrong'}">${scored.correct ? '✓' : '✗'} ${myPick.tip}</span>`;
          ptsEl = `<div class="result-pts">+${scored.pts}</div>`;
        }
      } else if (myPick) {
        tag = `<span class="result-pick-tag pending">⏳ ${myPick.tip}</span>`;
      } else {
        tag = `<span class="result-pick-tag pending">— No tip</span>`;
      }
      return `<div class="result-fixture">
        <div>
          <div class="result-teams">${f.home} vs ${f.away}</div>
          ${result?.winner ? `<div class="result-winner">✓ WINNER: ${result.winner}${result.margin ? ` by ${result.margin} pts` : ''}</div>` : '<div class="result-winner" style="color:var(--muted)">Result pending</div>'}
        </div>
        <div style="display:flex;align-items:center;gap:12px">${tag}${ptsEl}</div>
      </div>`;
    }).join('');

    if (total) {
      document.getElementById('results-my-summary').style.display = 'block';
      document.getElementById('res-pts').textContent     = Math.round(pts * 10) / 10;
      document.getElementById('res-correct').textContent = correct;
      document.getElementById('res-rate').textContent    = Math.round((correct / total) * 100) + '%';
    }
    empty.style.display = fixturesR.length ? 'none' : 'block';
  } catch { list.innerHTML = ''; empty.style.display = 'block'; }
}

// Client-side score helper (mirrors server)
function scoreOnePick(pick, result) {
  if (!result || result.status !== 'final') return null;
  let pts = 0;
  const correct = pick.tip === result.winner;
  if (correct) {
    pts += 1;
    if (pick.confidence) pts += Math.round((pick.confidence / 8) * 4 * 10) / 10;
    if (pick.margin != null && result.margin != null) {
      const diff = Math.abs(pick.margin - result.margin);
      if (diff === 0)      pts += 5;
      else if (diff <= 6)  pts += 2;
      else if (diff <= 12) pts += 1;
    }
  }
  if (pick.banker) pts *= 2;
  return { pts: Math.round(pts * 10) / 10, correct };
}

async function createComp() {
  const name  = document.getElementById('comp-name').value.trim();
  const sport = document.getElementById('comp-sport').value;
  if (!name) return showToast('Enter a comp name');
  try {
    const r = await fetch('/api/tipping/comps/create', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-session-token': tipToken },
      body: JSON.stringify({ name, sport })
    });
    const d = await r.json();
    if (d.success) {
      showToast(`✓ Comp created! Code: ${d.code}`);
      document.getElementById('comp-name').value = '';
      loadComps();
    } else showToast('⚠️ ' + (d.error || 'Failed'));
  } catch { showToast('⚠️ Connection error'); }
}

async function joinComp() {
  const code = document.getElementById('join-code').value.trim().toUpperCase();
  if (!code) return showToast('Enter an invite code');
  try {
    const r = await fetch('/api/tipping/comps/join', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', 'x-session-token': tipToken },
      body: JSON.stringify({ code })
    });
    const d = await r.json();
    if (d.success) {

      loadComps();

}
